package main

import (
	"os"

	"github.com/dimkarp93/install-libs/buildinfo"
)

var (
	version  string
	origin   string
	upstream string
	commit   string
	channel  string
)

func build() buildinfo.Info {
	return buildinfo.Info{
		Version:  version,
		Origin:   origin,
		Upstream: upstream,
		Commit:   commit,
		Channel:  channel,
	}
}

func main() {
	if build().Handle(os.Args[1:]) {
		return
	}
	run()
}
